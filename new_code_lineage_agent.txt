        prompt = ChatPromptTemplate.from_messages([
            SystemMessage(content=SYSTEM_PROMPT),
            MessagesPlaceholder(variable_name="messages")
        ])

        messages = prompt.format_messages(messages=state["messages"])

        response = model_with_tools.invoke(messages)

        return {"messages": [response]}

    tool_node = ToolNode(tools)

    builder = StateGraph(State)

    builder.add_node("chatbot", chatbot)
    builder.add_node("tools", tool_node)

    builder.add_edge(START, "chatbot")

    builder.add_conditional_edges("chatbot", tools_condition)

    builder.add_edge("tools", "chatbot")

    builder.add_edge("chatbot", END)

    graph = builder.compile(checkpointer=memory)

    return graph

import re

# -------------------------
# Read ETL file
# -------------------------

def read_etl_file(file_path):

    with open(file_path, "r", encoding="utf-8") as f:
        return f.read()

# -------------------------
# Smart stage splitter
# -------------------------

STAGE_PATTERNS = [
    r"--- \[.*?: .*?\]",
    r"Stage\s*:\s*.*",
    r"Component\s*:\s*.*",
    r"Node\s*:\s*.*"
]

def smart_stage_splitter(text):

    blocks = []

    pattern = "|".join(STAGE_PATTERNS)

    matches = list(re.finditer(pattern, text))

    for i, m in enumerate(matches):

        start = m.start()
        end = matches[i + 1].start() if i + 1 < len(matches) else len(text)

        blocks.append(text[start:end])

    return blocks


# -------------------------
# Structure extractor
# -------------------------

INPUT_KEYS = ["Input", "Source", "From"]
OUTPUT_KEYS = ["Output", "Target", "Write"]

def structure_extractor(block):

    stage = None
    inputs = []
    outputs = []

    stage_match = re.search(
        r"\[(?:CUSTOMSTAGE|TRANSFORMERSTAGE|HASHEDFILESTAGE|SEQFILESTAGE)\s*:\s*(.*?)\]",
        block
    )

    if stage_match:
        stage = stage_match.group(1).strip()

    for key in INPUT_KEYS:

        pattern = rf"{key}\s*:\s*.*?\((.*?)\)"
        matches = re.findall(pattern, block)

        inputs.extend(matches)

    for key in OUTPUT_KEYS:

        pattern = rf"{key}\s*:\s*.*?\((.*?)\)"
        matches = re.findall(pattern, block)

        outputs.extend(matches)

    return {
        "stage": stage,
        "inputs": inputs,
        "outputs": outputs
    }


# -------------------------
# Semantic extraction
# -------------------------

def extract_lineage_semantics(block):

    semantics = {}

    semantics["stage_variables"] = re.findall(r"StageVar\s+(\w+)", block)

    semantics["constraints"] = re.findall(r"Constraint\s*\((.*?)\)", block)

    semantics["joins"] = re.findall(
        r"JOIN\s+([A-Za-z0-9_.]+)", block, re.IGNORECASE
    )

    semantics["transformations"] = re.findall(r"=\s*(.*)", block)

    return semantics


# -------------------------
# Lineage graph builder
# -------------------------

def lineage_graph_builder(structure_data):

    nodes = []
    edges = []

    seen = set()

    for item in structure_data:

        stage = item["stage"]

        if not stage:
            continue

        if stage not in seen:
            nodes.append({"name": stage})
            seen.add(stage)

        for inp in item["inputs"]:

            if inp not in seen:
                nodes.append({"name": inp})
                seen.add(inp)

            if inp != stage:
                edges.append({
                    "source": inp,
                    "target": stage,
                    "label": "input"
                })

        for out in item["outputs"]:

            if out not in seen:
                nodes.append({"name": out})
                seen.add(out)

            if out != stage:
                edges.append({
                    "source": stage,
                    "target": out,
                    "label": "output"
                })

    return nodes, edges

import uuid
import networkx as nx

def layered_layout(G):

    try:
        layers = list(nx.topological_generations(G))
    except:
        layers = [list(G.nodes)]

    pos = {}

    x_gap = 350
    y_gap = 140

    for i, layer in enumerate(layers):

        for j, node in enumerate(layer):

            pos[node] = (i * x_gap, j * y_gap)

    return pos


def generate_drawio(nodes, edges):

    G = nx.DiGraph()

    for n in nodes:
        G.add_node(n["name"])

    for e in edges:
        G.add_edge(e["source"], e["target"])

    pos = layered_layout(G)

    node_ids = {}

    xml = []

    xml.append("<mxfile><diagram><mxGraphModel><root>")
    xml.append('<mxCell id="0"/><mxCell id="1" parent="0"/>')

    for node in nodes:

        node_id = str(uuid.uuid4())

        node_ids[node["name"]] = node_id

        x, y = pos.get(node["name"], (0, 0))

        xml.append(
            f'<mxCell id="{node_id}" value="{node["name"]}" '
            f'style="rounded=1;whiteSpace=wrap;html=1;" '
            f'vertex="1" parent="1">'
            f'<mxGeometry x="{x}" y="{y}" width="220" height="80" as="geometry"/>'
            '</mxCell>'
        )

    for e in edges:

        xml.append(
            f'<mxCell edge="1" parent="1" '
            f'source="{node_ids[e["source"]]}" '
            f'target="{node_ids[e["target"]]}" '
            f'value="{e["label"]}">'
            '<mxGeometry relative="1" as="geometry"/>'
            '</mxCell>'
        )

    xml.append("</root></mxGraphModel></diagram></mxfile>")

    return "\n".join(xml)

import asyncio

from agent_template.lineage_engine import build_agent

load_dotenv()

async def run_agent():

    graph = await build_agent()

    config = {
        "configurable": {"thread_id": "etl-session"},
        "recursion_limit": 100
    }

    print("ETL Lineage Agent Ready")

    while True:

        user_input = input("Enter command: ")

        if user_input.lower() in ["exit", "quit"]:
            break

        events = graph.astream(
            {"messages": [("user", user_input)]},
            config=config
        )

        async for event in events:

            for value in event.values():

                if "messages" in value:

                    print("\nAgent:", value["messages"][-1].content)


if __name__ == "__main__":
    asyncio.run(run_agent())
