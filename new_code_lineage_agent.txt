import uuid
import networkx as nx


# ------------------------------------------------
# Layout builder (cycle safe + ETL friendly)
# ------------------------------------------------

def layered_layout(G):

    pos = {}

    try:
        # Works when graph is DAG
        layers = list(nx.topological_generations(G))
    except:
        # fallback for cyclic graphs
        layers = [list(G.nodes)]

    x_gap = 350
    y_gap = 140

    for i, layer in enumerate(layers):

        for j, node in enumerate(layer):

            pos[node] = (i * x_gap, j * y_gap)

    return pos


# ------------------------------------------------
# Node styling based on type
# ------------------------------------------------

def get_node_style(node):

    node_type = node.get("type", "stage")

    if node_type == "source":
        return "rounded=1;whiteSpace=wrap;html=1;fillColor=#d5e8d4"

    if node_type == "dataset":
        return "rounded=1;whiteSpace=wrap;html=1;fillColor=#fff2cc"

    if node_type == "stage":
        return "rounded=1;whiteSpace=wrap;html=1;fillColor=#dae8fc"

    if node_type == "target":
        return "rounded=1;whiteSpace=wrap;html=1;fillColor=#f8cecc"

    return "rounded=1;whiteSpace=wrap;html=1"


# ------------------------------------------------
# Draw.io generator
# ------------------------------------------------

def generate_drawio(nodes, edges):

    G = nx.DiGraph()

    for n in nodes:
        G.add_node(n["name"])

    for e in edges:
        G.add_edge(e["source"], e["target"])

    pos = layered_layout(G)

    node_ids = {}

    xml = []

    xml.append("<mxfile><diagram><mxGraphModel><root>")
    xml.append('<mxCell id="0"/><mxCell id="1" parent="0"/>')

    # ----------------------------
    # create nodes
    # ----------------------------

    for node in nodes:

        node_name = node["name"]

        node_id = str(uuid.uuid4())

        node_ids[node_name] = node_id

        x, y = pos.get(node_name, (0, 0))

        style = get_node_style(node)

        xml.append(
            f'<mxCell id="{node_id}" value="{node_name}" '
            f'style="{style}" vertex="1" parent="1">'
            f'<mxGeometry x="{x}" y="{y}" width="220" height="80" as="geometry"/>'
            '</mxCell>'
        )

    # ----------------------------
    # create edges
    # ----------------------------

    for e in edges:

        source = e["source"]
        target = e["target"]

        if source not in node_ids or target not in node_ids:
            continue

        label = e.get("label", "")

        xml.append(
            f'<mxCell edge="1" parent="1" '
            f'source="{node_ids[source]}" '
            f'target="{node_ids[target]}" '
            f'value="{label}">'
            '<mxGeometry relative="1" as="geometry"/>'
            '</mxCell>'
        )

    xml.append("</root></mxGraphModel></diagram></mxfile>")

    return "\n".join(xml)
