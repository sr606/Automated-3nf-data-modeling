import re


def read_etl_file(path):

    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def smart_stage_splitter(text):

    print("\n[TOOL] smart_stage_splitter running")

    blocks = []

    pattern = r"--- \[.*?\]"

    matches = list(re.finditer(pattern, text))

    for i, m in enumerate(matches):

        start = m.start()
        end = matches[i + 1].start() if i + 1 < len(matches) else len(text)

        block = text[start:end]

        blocks.append(block)

    print(f"[TOOL] Found {len(blocks)} stage blocks")

    return blocks


def lineage_graph_builder(metadata):

    print("\n[TOOL] lineage_graph_builder running")

    nodes = []
    edges = []

    seen_nodes = set()

    for item in metadata:

        stage = item["stage"]
        inputs = item["inputs"]
        outputs = item["outputs"]

        if stage not in seen_nodes:
            nodes.append({"name": stage, "type": "stage"})
            seen_nodes.add(stage)

        for inp in inputs:

            if inp not in seen_nodes:
                nodes.append({"name": inp, "type": "dataset"})
                seen_nodes.add(inp)

            edges.append({
                "source": inp,
                "target": stage,
                "label": "input"
            })

        for out in outputs:

            if out not in seen_nodes:
                nodes.append({"name": out, "type": "dataset"})
                seen_nodes.add(out)

            edges.append({
                "source": stage,
                "target": out,
                "label": "output"
            })

    print("[TOOL] Graph built")
    print("Nodes:", nodes)
    print("Edges:", edges)

    return nodes, edges
4️⃣ diagram_generator.py (clean layout)

lineage_mcp/tools/diagram_generator.py

import uuid
import networkx as nx


def layered_layout(G):

    pos = {}

    try:
        layers = list(nx.topological_generations(G))
    except:
        layers = [list(G.nodes)]

    x_gap = 350
    y_gap = 140

    for i, layer in enumerate(layers):

        for j, node in enumerate(layer):

            pos[node] = (i * x_gap, j * y_gap)

    return pos


def node_style(node):

    t = node.get("type")

    if t == "stage":
        return "rounded=1;whiteSpace=wrap;html=1;fillColor=#dae8fc"

    if t == "dataset":
        return "rounded=1;whiteSpace=wrap;html=1;fillColor=#fff2cc"

    return "rounded=1;whiteSpace=wrap;html=1"


def generate_drawio(nodes, edges):

    print("\n[TOOL] generate_drawio running")

    G = nx.DiGraph()

    for n in nodes:
        G.add_node(n["name"])

    for e in edges:
        G.add_edge(e["source"], e["target"])

    pos = layered_layout(G)

    node_ids = {}

    xml = []

    xml.append("<mxfile><diagram><mxGraphModel><root>")
    xml.append('<mxCell id="0"/><mxCell id="1" parent="0"/>')

    for node in nodes:

        nid = str(uuid.uuid4())
        node_ids[node["name"]] = nid

        x, y = pos[node["name"]]

        style = node_style(node)

        xml.append(
            f'<mxCell id="{nid}" value="{node["name"]}" '
            f'style="{style}" vertex="1" parent="1">'
            f'<mxGeometry x="{x}" y="{y}" width="220" height="80" as="geometry"/>'
            '</mxCell>'
        )

    for e in edges:

        xml.append(
            f'<mxCell edge="1" parent="1" '
            f'source="{node_ids[e["source"]]}" '
            f'target="{node_ids[e["target"]]}" '
            f'value="{e["label"]}">'
            '<mxGeometry relative="1" as="geometry"/>'
            '</mxCell>'
        )

    xml.append("</root></mxGraphModel></diagram></mxfile>")

    print("[TOOL] Drawio XML generated")

    return "\n".join(xml)

You are an Enterprise ETL Lineage Extraction Agent.

You are connected to deterministic lineage tools through MCP.

Your responsibility is to extract stage metadata and build a lineage graph.

STRICT RULES:

1. NEVER invent datasets or stages.
2. ONLY extract information visible in the ETL pseudocode.
3. Stage metadata must follow this JSON schema:

{
 "stage": string,
 "inputs": [string],
 "outputs": [string],
 "stage_type": string
}

4. Inputs are datasets consumed by the stage.
5. Outputs are datasets produced by the stage.

WORKFLOW:

1. read_etl_file
2. smart_stage_splitter
3. For each block:
   extract stage metadata
4. Build metadata list
5. call lineage_graph_builder
6. call generate_drawio

Return structured metadata only.

Never hallucinate lineage relationships.
7️⃣ main.py
import asyncio
from agent_template.lineage_engine import build_agent


async def run():

    graph = await build_agent()

    config = {
        "configurable": {"thread_id": "etl-session"}
    }

    print("ETL Lineage Agent Ready")

    while True:

        user = input("Enter command: ")

        events = graph.astream(
            {"messages": [("user", user)]},
            config=config
        )

        async for event in events:

            for v in event.values():

                if "messages" in v:
                    print("\nAgent:", v["messages"][-1].content)


if __name__ == "__main__":
    asyncio.run(run())
