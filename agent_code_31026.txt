import uuid
import networkx as nx
 
 
# ------------------------------------------------
# Layout
# ------------------------------------------------
 
def layered_layout(G):
 
    pos = {}
 
    try:
        layers = list(nx.topological_generations(G))
    except:
        layers = [list(G.nodes)]
 
    x_gap = 300
    y_gap = 180
 
    for i, layer in enumerate(layers):
        for j, node in enumerate(layer):
            pos[node] = (i * x_gap, j * y_gap)
 
    return pos
 
 
# ------------------------------------------------
# Node styling
# ------------------------------------------------
 
def node_style(node):
 
    t = node.get("type")
 
    if t == "stage":
        return "rounded=1;whiteSpace=wrap;html=1;fillColor=#dae8fc;"
 
    if t == "dataset":
        return "rounded=1;whiteSpace=wrap;html=1;fillColor=#fff2cc;"
 
    if t == "exception":
        return "rounded=1;whiteSpace=wrap;html=1;fillColor=#ffe6cc;"
 
    return "rounded=1;whiteSpace=wrap;html=1;"
 
 
# ------------------------------------------------
# Draw.io generator
# ------------------------------------------------
 
def generate_drawio(nodes, edges):
 
    print("\n[TOOL] generate_drawio running")
 
    G = nx.DiGraph()
 
    for n in nodes:
        G.add_node(n["name"])
 
    for e in edges:
        G.add_edge(e["source"], e["target"])
 
    pos = layered_layout(G)
 
    node_ids = {}
    xml = []
 
    xml.append("<mxfile>")
    xml.append("<diagram>")
    xml.append("<mxGraphModel>")
    xml.append("<root>")
    xml.append('<mxCell id="0"/>')
    xml.append('<mxCell id="1" parent="0"/>')
 
    # -------------------------
    # Nodes
    # -------------------------
 
    for node in nodes:
 
        nid = str(uuid.uuid4())
        node_ids[node["name"]] = nid
 
        x, y = pos[node["name"]]
 
        style = node_style(node)
 
        name = node["name"]
 
        label = name
 
        xml.append(
            f'<mxCell id="{nid}" value="{label}" style="{style}" vertex="1" parent="1">'
        )
 
        xml.append(
            f'<mxGeometry x="{x}" y="{y}" width="220" height="80" as="geometry"/>'
        )
 
        xml.append('</mxCell>')
 
    # -------------------------
    # Edges
    # -------------------------
 
    for e in edges:
 
        xml.append(
            f'<mxCell edge="1" parent="1" '
            f'source="{node_ids[e["source"]]}" '
            f'target="{node_ids[e["target"]]}" '
            f'value="{e["label"]}">'
        )
 
        xml.append('<mxGeometry relative="1" as="geometry"/>')
        xml.append('</mxCell>')
 
    xml.append("</root>")
    xml.append("</mxGraphModel>")
    xml.append("</diagram>")
    xml.append("</mxfile>")
 
    print("[TOOL] Drawio XML generated")
 
    return "\n".join(xml)

import os
import re
import networkx as nx
 
 
# ------------------------------------------------
# Read ETL file
# ------------------------------------------------
 
def read_etl_file(path):
 
    with open(path, "r", encoding="utf-8") as f:
        return f.read()
 
 
# ------------------------------------------------
# List upload files
# ------------------------------------------------
 
def list_upload_files():
 
    folder = "./lineage_mcp/data/upload"
 
    if not os.path.exists(folder):
        return []
 
    files = [
        f for f in os.listdir(folder)
        if f.endswith(".txt")
    ]
 
    return files
 
 
# ------------------------------------------------
# Stage splitter
# ------------------------------------------------
 
def smart_stage_splitter(text):
 
    print("\n[TOOL] smart_stage_splitter")
 
    pattern = r"---\s*\[[^\]]+\]"
 
    matches = list(re.finditer(pattern, text))
 
    blocks = []
 
    for i, m in enumerate(matches):
 
        start = m.start()
 
        end = matches[i + 1].start() if i + 1 < len(matches) else len(text)
 
        blocks.append(text[start:end])
 
    print(f"[TOOL] Total stage blocks detected: {len(blocks)}")
 
    return blocks
 
 
# ------------------------------------------------
# Logical chunking (LLM safe)
# ------------------------------------------------
 
def logical_chunk_stage(block, max_chars=3000):
 
    sections = re.split(
        r"(SQL:|Stage Variables:|Transformations:|Constraint)",
        block
    )
 
    chunks = []
 
    current = ""
 
    for part in sections:
 
        if len(current) + len(part) < max_chars:
            current += part
        else:
            chunks.append(current)
            current = part
 
    if current:
        chunks.append(current)
 
    return chunks
 

def lineage_graph_builder(metadata):
 
    print("\n[TOOL] lineage_graph_builder")
 
    nodes = []
    edges = []
 
    seen_nodes = set()
    seen_edges = set()
 
    def add_node(name, node_type="dataset"):
 
        if name not in seen_nodes:
            nodes.append({
                "name": name,
                "type": node_type
            })
            seen_nodes.add(name)
 
    def add_edge(src, tgt, label):
 
        key = (src, tgt, label)
 
        if key not in seen_edges:
            edges.append({
                "source": src,
                "target": tgt,
                "label": label
            })
            seen_edges.add(key)
 
    for item in metadata:
 
        inputs = item.get("inputs", [])
        lookups = item.get("lookups", [])
        outputs = item.get("outputs", [])
        exceptions = item.get("exceptions", [])
 
        # register nodes
        for n in inputs + lookups + outputs + exceptions:
            add_node(n)
 
        # input → output
        for inp in inputs:
            for out in outputs:
                add_edge(inp, out, "input")
 
        # lookup → output
        for lk in lookups:
            for out in outputs:
                add_edge(lk, out, "lookup")
 
        # input → exception
        for inp in inputs:
            for ex in exceptions:
                add_edge(inp, ex, "exception")
 
    print("[TOOL] Nodes:", nodes)
    print("[TOOL] Edges:", edges)
 
    return nodes, edges
 
 
# ------------------------------------------------
# Graph correction layer
# ------------------------------------------------
 
def correct_lineage_graph(nodes, edges):
 
    print("\n[GRAPH] Running graph correction layer")
 
    G = nx.DiGraph()
 
    for n in nodes:
        G.add_node(n["name"], type=n.get("type", "dataset"))
 
    for e in edges:
 
        src = e["source"]
        tgt = e["target"]
        label = e.get("label", "flow")
 
        if src == tgt:
            continue
 
        G.add_edge(src, tgt, label=label)
 
    # Remove cycles
    try:
 
        cycles = list(nx.simple_cycles(G))
 
        for cycle in cycles:
 
            print("[GRAPH] Cycle detected:", cycle)
 
            G.remove_edge(cycle[-1], cycle[0])
 
    except Exception:
        pass
 
    corrected_edges = []
 
    for u, v, data in G.edges(data=True):
 
        corrected_edges.append({
            "source": u,
            "target": v,
            "label": data.get("label", "flow")
        })
 
    corrected_nodes = []
 
    for n, data in G.nodes(data=True):
 
        corrected_nodes.append({
            "name": n,
            "type": data.get("type", "dataset")
        })
 
    print("[GRAPH] Graph correction complete")
 
    return corrected_nodes, corrected_edges

# ------------------------------------------------
# MCP output normalization
# ------------------------------------------------
 
def normalize_mcp_output(resp):
 
    # MCP commonly returns list of text blocks
    if isinstance(resp, list):
 
        texts = []
 
        for item in resp:
            if isinstance(item, dict) and "text" in item:
                texts.append(item["text"])
 
        if texts:
            if len(texts) == 1:
                return texts[0]
            return texts
 
        return resp
 
    # dict response
    if isinstance(resp, dict):
 
        if "text" in resp:
            return resp["text"]
 
        return resp
 
    # JSON string
    if isinstance(resp, str):
 
        try:
            return json.loads(resp)
        except:
            return resp
 
    return resp
 
 
# ------------------------------------------------
# Safe JSON parser
# ------------------------------------------------
 
def safe_json_parse(text):
 
    if not text:
        raise ValueError("LLM returned empty response")
 
    text = text.strip()
 
    # remove markdown code blocks if present
    if text.startswith("```"):
        parts = text.split("```")
        if len(parts) >= 2:
            text = parts[1].strip()
 
    # extract JSON portion if extra text exists
    start = text.find("{")
    end = text.rfind("}")
 
    if start != -1 and end != -1:
        text = text[start:end + 1]
 
    try:
        return json.loads(text)
 
    except json.JSONDecodeError:
        print("\n[WARN] Invalid JSON returned by LLM:")
        print(text)
        raise

# ------------------------------------------------
# Build agent
# ------------------------------------------------
 
async def build_agent():
 
    tools = await get_tools()
 
    tool_map = {t.name: t for t in tools}
 
    read_tool = tool_map["read_etl_file_tool"]
    split_tool = tool_map["smart_stage_splitter_tool"]
    graph_tool = tool_map["lineage_graph_builder_tool"]
    draw_tool = tool_map["generate_drawio_tool"]
 
 
    async def workflow(state: State):
 
        user_input = state["messages"][-1].content
        print("\n[AGENT] Request:", user_input)
 
        # ------------------------------------------------
        # Detect files
        # ------------------------------------------------
 
        if "upload folder" in user_input.lower():
            files = list_upload_files()
 
        else:
 
            match = re.search(r'for\s+(.+\.txt)', user_input)
 
            if not match:
                raise ValueError("File name not detected")
 
            files = [match.group(1)]
 
        print("[AGENT] Files:", files)
 
        # ------------------------------------------------
        # Process files
        # ------------------------------------------------
 
        for file_name in files:
 
            print("\n[AGENT] Processing:", file_name)
 
            file_resp = await read_tool.ainvoke({
                "file_name": file_name
            })
 
            text = normalize_mcp_output(file_resp)
 
            if isinstance(text, list):
                text = "\n".join(text)
 
            if not isinstance(text, str):
                raise ValueError(f"Invalid text type: {type(text)}")
 
            # ------------------------------------------------
            # Stage splitting
            # ------------------------------------------------
 
            split_resp = await split_tool.ainvoke({
                "text": text
            })
 
            blocks = normalize_mcp_output(split_resp)
 
            if not isinstance(blocks, list):
                raise ValueError(f"Stage splitter returned invalid type {type(blocks)}")
 
            print("\n[AGENT] Total stage blocks:", len(blocks))
            print("\n[DEBUG] First stage block preview:\n")
            print(blocks[0][:500])
 
            metadata = []
 
            # ------------------------------------------------
            # Process blocks
            # ------------------------------------------------
 
            for block in blocks:
 
                chunks = logical_chunk_stage(block)
 
                stage_header = block.split("\n")[0]
 
                combined = None
 
                for chunk in chunks:
 
                    chunk_input = stage_header + "\n" + chunk
 
                    messages = [
                        SystemMessage(content=SYSTEM_PROMPT),
                        HumanMessage(content=chunk_input)
                    ]
 
                    response = await llm.ainvoke(messages)

                    try:
                        data = safe_json_parse(response.content)

                    except Exception:
                        print("[WARN] Skipping invalid LLM response")
                        continue
 
                    print("[LLM]", data)
 
                    if not data["stage"]:
                        continue
 
                    if not combined:
                        combined = data
                    else:
                        combined["inputs"].extend(data["inputs"])
                        combined["outputs"].extend(data["outputs"])
 
                if combined:
                    metadata.append(combined)
 
            # ------------------------------------------------
            # Build lineage graph
            # ------------------------------------------------

            print("\n[DEBUG] Final merged metadat:")
            for m in metadata:
                print(m)
 
            graph_resp = await graph_tool.ainvoke({
                "metadata": metadata
            })
 
            graph = normalize_mcp_output(graph_resp)
 
            if isinstance(graph, str):
                graph = json.loads(graph)
 
            if not isinstance(graph, dict):
                raise ValueError(f"Unexpected graph output: {type(graph)}")
 
            nodes = graph["nodes"]
            edges = graph["edges"]

            print("\n[DEBUG] Edges builts:")
            for e in edges:
                print(e)
 
            # ------------------------------------------------
            # Generate draw.io diagram
            # ------------------------------------------------
 
            draw_resp = await draw_tool.ainvoke({
                "nodes": nodes,
                "edges": edges,
                "file_name": file_name.replace(".txt", "")
            })
 
            normalize_mcp_output(draw_resp)
 
        return {
            "messages": [
                HumanMessage(content="Lineage generation complete.")
            ]
        }
 
    builder = StateGraph(State)
 
    builder.add_node("workflow", workflow)
 
    builder.add_edge(START, "workflow")
    builder.add_edge("workflow", END)
 
    graph = builder.compile(checkpointer=memory)
 
    return graph
