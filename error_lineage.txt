@router.post(
    "/available_lineage_sources",
    operation_id="available_lineage_sources",
    summary="Returns ETL pseudocode files available for lineage generation",
    description="Scans lineage source directory and returns ETL pseudocode files available for lineage analysis."
)
async def available_lineage_sources(p_body: AvailableLineageSources):
    """
    Retrieve ETL pseudocode files available for lineage processing.

    This endpoint scans the lineage source directory associated with the user
    and returns pseudocode files that can be used as input for lineage extraction.

    These files typically contain ETL job pseudocode exported from ETL tools
    such as DataStage or similar systems.

    NOTE: Steps to follow:
        1. Call this endpoint first to identify available lineage input files.
        2. Select the required pseudocode files.
        3. Pass selected files to 'read_lineage_sources' tool.

    Args:
        p_body (AvailableLineageSources):
            user_id (str): Unique identifier for the user.

    Returns:
        dict:
            Available_Lineage_Sources (List[str]): List of available ETL pseudocode files.
            error (str, optional): Error message if retrieval fails.
    """

    try:

        log_file = methods.get_or_create_file(
            agent_name="lineage_agent",
            tool_name="available_lineage_sources"
        )

        execution_id = datetime.now().strftime('%Y-%m-%d-%H-%M-%S')

        lineage_dir = os.path.join(
            os.getenv("UPLOAD_BASE_PATH"),
            f"lineage/{p_body.user_id}/sources/"
        )

        files = methods.list_files_in_directory(lineage_dir)

        dbg.dprint2(script_name, execution_id, "status", "Lineage sources fetched", log_file)

        return {"Available_Lineage_Sources": files}

    except Exception as e:
        tb = traceback.format_exc()
        dbg.dprint2(script_name, execution_id, "exception", tb, log_file)
        return {"error": str(e)}

@router.post(

    description="Reads ETL pseudocode files and consolidates them into a lineage source JSON artifact."
)
async def read_lineage_sources(p_body: ReadLineageSources):
    """
    Load pseudocode ETL job files.

    This endpoint reads one or more pseudocode ETL job files and consolidates them
    into a structured lineage source JSON file used by downstream lineage tools.

    NOTE: Steps to follow:
        1. Use 'available_lineage_sources' tool to identify available pseudocode files.
        2. Select required ETL pseudocode files.
        3. Call this endpoint to generate a lineage source JSON file.

    Returns:
        dict:
            Lineage_Source_File (str): Generated lineage source file.
    """

    try:

        execution_id = datetime.now().strftime('%Y%m%d%H%M%S')

        filename = f"{execution_id}_lineage_sources.json"

        source_dir = os.path.join(
            os.getenv("UPLOAD_BASE_PATH"),
            f"lineage/{p_body.user_id}/sources/"
        )

        os.makedirs(source_dir, exist_ok=True)

        file_path = os.path.join(source_dir, filename)

        job_paths = [
            os.path.join(source_dir, f)
            for f in p_body.job_files
        ]

        methods.read_etl_sources(job_paths, file_path)

        methods.save_lineage_state(
            execution_id,
            {"source_file": filename, "source_files": p_body.job_files}
        )

        return {"Lineage_Source_File": filename}

    except Exception as e:
        return {"error": str(e)}
